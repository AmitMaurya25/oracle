Create Policy

SQL> conn sys/admin@cdblkopdbhr as sysdba
Connected.


SQL> CREATE AUDIT POLICY ram_audit_policy
    PRIVILEGES
    CREATE TABLE, CREATE SEQUENCE
    WHEN 'SYS_CONTEXT(''USERENV'', ''SESSION_USER'') = ''RAM'''
    EVALUATE PER SESSION
    CONTAINER = CURRENT;

Audit policy created.



Enable the auditing policy.

AUDIT POLICY ram_audit_policy;
